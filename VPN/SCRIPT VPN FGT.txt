 config vpn ipsec phase1-interface
    edit "TO_SITE02"
        set interface "port1"
        set ike-version 2
        set keylife 28800
        set peertype any
        set net-device disable
        set proposal des-sha1
        set dhgrp 14
        set remote-gw 172.17.174.179
        set psksecret teste0102
    next
end

config vpn ipsec phase2-interface
    edit "TO_SITE02"
        set phase1name "TO_SITE02"
        set proposal des-sha1
        set dhgrp 14
        set auto-negotiate enable
        set keylifeseconds 3600
        set src-subnet 10.0.0.0 255.255.255.0
	set dst-subnet 0.0.0.0 0.0.0.0
    next
end
config system interface

        set ip 169.255.1.2 255.255.255.255
        set allowaccess ping
        set type tunnel
        set remote-ip 169.255.1.1 255.255.255.252

end

    next
end
